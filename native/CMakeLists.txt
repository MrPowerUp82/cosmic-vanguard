cmake_minimum_required(VERSION 3.20)
project(CosmicVanguardNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(WIN32)
  find_package(SDL2 REQUIRED CONFIG)
  find_package(SDL2_image REQUIRED CONFIG)
  find_package(SDL2_ttf REQUIRED CONFIG)
  set(COSMIC_SDL SDL2::SDL2main SDL2::SDL2 SDL2_image::SDL2_image SDL2_ttf::SDL2_ttf)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf)
  set(COSMIC_SDL PkgConfig::SDL2)
endif()
add_library(cosmic_core src/game.cpp)
target_include_directories(cosmic_core PUBLIC include)
target_compile_options(cosmic_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(cosmic_vanguard src/main.cpp)
target_link_libraries(cosmic_vanguard PRIVATE cosmic_core ${COSMIC_SDL})
if(MINGW)
  set_target_properties(cosmic_vanguard PROPERTIES WIN32_EXECUTABLE ON)
  target_link_options(cosmic_vanguard PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic,--whole-archive -lwinpthread -Wl,--no-whole-archive,-Bdynamic)
endif()
enable_testing()
add_executable(cosmic_core_tests tests/core_tests.cpp)
target_link_libraries(cosmic_core_tests PRIVATE cosmic_core)
add_test(NAME cosmic_core_tests COMMAND cosmic_core_tests)
