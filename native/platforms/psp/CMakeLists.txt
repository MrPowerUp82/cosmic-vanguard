cmake_minimum_required(VERSION 3.20)
project(CosmicVanguardPsp LANGUAGES C CXX)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
find_package(SDL2 REQUIRED)
add_executable(cosmic_psp ${ROOT}/src/main.cpp ${ROOT}/src/game.cpp)
target_include_directories(cosmic_psp PRIVATE ${ROOT}/include)
target_compile_features(cosmic_psp PRIVATE cxx_std_17)
target_compile_definitions(cosmic_psp PRIVATE COSMIC_PSP)
target_compile_options(cosmic_psp PRIVATE -O2 -G0 -fsingle-precision-constant)
target_link_libraries(cosmic_psp PRIVATE SDL2_ttf SDL2_image freetype harfbuzz png jpeg z bz2 SDL2::SDL2main SDL2::SDL2-static)
create_pbp_file(TARGET cosmic_psp TITLE "Cosmic Vanguard" ICON_PATH ${CMAKE_CURRENT_LIST_DIR}/icon0.png BACKGROUND_PATH ${CMAKE_CURRENT_LIST_DIR}/pic1.png VERSION 00.10 BUILD_PRX)
